% ====== PART I: PROBLEM SETTING ======

\begin{frame}{Motivation \& Context}
\begin{itemize}
  \item Compute selected eigenpairs of large matrices: $\mat{A}\vec{x} = \lambda\vec{x}$
  \item \textbf{Challenges:} scalability, memory, communication,
        clustered/interior eigenvalues, conditioning, preconditioning, \ldots
\end{itemize}
\pause
\textbf{Dominant costs:}
\begin{itemize}
  \item Matvecs with $\mat{A}$
  \item \only<2>{Orthogonalization of search space $\mat{V}$}%
        \only<3>{\rr{Orthogonalization of search space $\mat{V}$}}
  \item Solving the reduced problem $\mat{V}^*\!\mat{AV}\vec{y} = \theta\vec{y}$
\end{itemize}
\pause
\begin{block}{Algorithms}
\begin{itemize}
  \item \textbf{Krylov-Schur}:
        {\footnotesize de Damas--Grigori (2025); Grigori--Kressner--Shao--Simunec (2026)}
  \item \rr{\textbf{JD} (Jacobi--Davidson)}: Rayleigh--Ritz + correction equation;
        good for interior/clustered eigenvalues.
  \item \textbf{LOBPCG} (Locally Optimal Block Preconditioned CG): efficient for
        Hermitian/SPD problems.
\end{itemize}
\end{block}
\end{frame}

% ====== PART II: CLASSICAL ORTHOGONALIZATION ======

\begin{frame}{Orthogonalizing a basis}
Given $\mat{V}=[\vec{v}_1,\ldots,\vec{v}_k]\in\R^{n\times k}$
with $\mat{V}^\top\mat{V}=\mat{I}$ and a new vector $\vec{t}\in\R^n$, find the
\emph{orthogonalized residual}:
\[
  \tilde{\vec{t}} = \vec{t} - \mat{V}\vec{h},
  \qquad
  \vec{h} = \argmin_{\vec{h}}\|\vec{t} - \mat{V}\vec{h}\|_2^2 = \mat{V}^\top\vec{t}.
\]
\emph{Same problem}, but differ in when they
apply the subtraction and how many communication rounds they use.

\pause
\vspace{0.4em}
\textbf{CGS2:}

\begin{tcolorbox}[colback=green!5,colframe=green!50!black,
  boxrule=0.4pt,top=2pt,bottom=2pt,left=3pt,right=3pt]
\begin{columns}[T]
\column{0.38\textwidth}
  
  \begin{algorithmic}[1]
    \State $\vec{h} \gets \mat{V}^\top \vec{t}$   
    \State $\tilde{\vec{t}} \gets \vec{t} - \mat{V}\vec{h}$
    \State $\vec{h}' \gets \mat{V}^\top \tilde{\vec{t}}$
    \State $\tilde{\vec{t}} \gets \tilde{\vec{t}} - \mat{V}\vec{h}'$
    \State $\vec{t} \gets \tilde{\vec{t}}/\|\tilde{\vec{t}}\|$
  \end{algorithmic}
\column{0.58\textwidth}
  \vspace{1em}
  \textbf{Projector:}
  \begin{equation*}
      \vec{\tilde{t}} \gets (\mat{I}-\mat{VV}^\top)(\mat{I}-\mat{VV}^\top)\vec{t}
  \end{equation*}
\end{columns}
\end{tcolorbox}
\end{frame}


% ====== PART IV: RANDNLA ======

\begin{frame}{RandNLA}
\begin{block}{Subspace embedding {\footnotesize [Sarl\'os, 2006]}}
A matrix $\mat{S}\in\R^{s\times m}$ is a subspace embedding for $\mat{A}\in\R^{m\times n}$
if for all $\vec{x}\in\operatorname{range}(\mat{A})$ and for some distortion $\eta\in(0,1)$,
\begin{equation*}
  (1-\eta)\norm{\vec{x}}_2^2 \;\leq\; \norm{\mat{S}\vec{x}}_2^2
  \;\leq\; (1+\eta)\norm{\vec{x}}_2^2.
\end{equation*}
Typically $s\ll m$ and $s=\bigO(\operatorname{rank}(\mat{A}))$.
\end{block}
\pause
\textbf{Examples:}
\begin{itemize}
  \item Scaled Gaussian embedding ($S_{ij}\sim\mathcal{N}(0,1/\sqrt{s})$)
  \item Subsampled Randomized Trigonometric Transforms (SRTTs)
  \item Sparse Sign Embedding
\end{itemize}
See also subspace injections {\footnotesize Cama\~{n}o--Epperly--Meyer--Tropp (2025)}
\end{frame}

\begin{frame}{Consequences}
\begin{itemize}
  \item Inner product is approximately preserved:
  \begin{equation*}
    |\langle\vec{x},\vec{y}\rangle - \langle\mat{S}\vec{x},\mat{S}\vec{y}\rangle|
    \;\leq\; \eta\norm{\vec{x}}_2\norm{\vec{y}}_2
  \end{equation*}
  for all $\vec{x},\vec{y}\in\operatorname{range}(\mat{A})$.
  \pause
  \item Singular values are approximately preserved:
  \begin{equation*}
    (1-\eta)\sigma_i(\mat{A})^2 \;\leq\;
    \sigma_i(\mat{SA})^2 \;\leq\; (1+\eta)\sigma_i(\mat{A})^2
  \end{equation*}
  \item Condition number is approximately preserved:
  \begin{equation*}
    \kappa_2(\mat{SA}) \;\leq\; \sqrt{\frac{1+\eta}{1-\eta}}\cdot\kappa_2(\mat{A}),
    \qquad \kappa_2(\mat{A})=\sigma_1(\mat{A})/\sigma_n(\mat{A}).
  \end{equation*}
\end{itemize}
\end{frame}

% ====== PART V: SKETCH-ORTHOGONALIZATION ======

% --- Transition: what the sketch buys ---
\begin{frame}{From embeddings to orthogonalization}
\textbf{Key connection:} if $\mat{S}$ is an $\eta$-embedding for
$\operatorname{span}(\mat{V}, \vec{t})$, then solving the \emph{sketched}
least-squares problem
\[
  \min_{\vec{h}}\|\mat{S}(\vec{t} - \mat{V}\vec{h})\|_2^2 = \min_{\vec{h}}\|\mat{S}\vec{t} - \mat{S}\mat{V}\vec{h}\|_2^2
\]
gives a residual within a factor $\sqrt{\frac{1+\eta}{1-\eta}} \approx 1$ of
the optimal $\ell_2$ residual. \\
\pause
\vspace{1em}
\textbf{RGS:} {\footnotesize [Balabanov, Grigori, 2022]} Keep $(\mat{SV})^\top (\mat{SV}) = \mat{I} \implies \kappa_2(\mat{V}) \leq \sqrt{\frac{1+\eta}{1-\eta}}$.
\begin{tcolorbox}[colback=orange!5,colframe=orange!60!black,
  boxrule=0.4pt,top=2pt,bottom=2pt,left=3pt,right=3pt]
\begin{columns}[T]
\column{0.50\textwidth}
  \begin{algorithmic}[1]
    \State $\vec{p} \gets \mat{S}\vec{t}$
    \State \rr{$\vec{h} \gets \mat{P}^\dagger \vec{p}$}
          \Comment{exact: $\mat{P}^\dagger = \mat{P}^\top$}
    \State $\tilde{\vec{t}} \gets \vec{t} {-} \mat{V}\vec{h}$;
           $\;\tilde{\vec{p}} \gets \vec{p} {-} \mat{P}\vec{h}$
    \State $\tau \gets \|\tilde{\vec{p}}\|$
    \State $\vec{t} \gets \tilde{\vec{t}}/\tau$;
           $\;\mat{P} \leftarrow [\mat{P},\, \tilde{\vec{p}}/\tau]$
  \end{algorithmic}
\column{0.46\textwidth}
  \footnotesize\vspace{1em}
  \textbf{RGS (Oblique) Projector:}
  \begin{equation*}
      \vec{\tilde{t}} \gets (\mat{I}-\mat{V}(\mat{SV})^\dagger\mat{S})\vec{t}
  \end{equation*}
\end{columns}
\end{tcolorbox}
\end{frame}

% ====== PART VI: JACOBI--DAVIDSON ======

\begin{frame}{Jacobi--Davidson algorithm (overview)}
 \textbf{Goal:} compute a few eigenpairs of
  $\mat{A}\in\R^{n\times n}$.
 
  \vspace{0.5em}
  \pause
 
  \textbf{Input:} $\mat{A}$, target shift $\sigma$,
  \alt<2>{initial vector $\vec{v}_1$}%
         {\bl{initial block $\mat{V}_0\in\R^{n\times b}$}}.
 
  \begin{enumerate}
 
    \item Build orthogonal subspace $\mat{V}_k=[\vec{v}_1,\ldots,\vec{v}_k]$

    \item Compute
          \alt<2>{Ritz pair $(\theta,\,\vec{u}=\mat{V}_k\vec{y})$}%
                 {\bl{Ritz block $(\mat{\Theta},\,\mat{U}=\mat{V}_k\mat{Y})$}}
          by solving
          \alt<2>{$\mat{V}_k^\top\mat{A}\mat{V}_k\vec{y}
                   =\theta\vec{y}$}%
                 {\bl{$\mat{V}_k^\top\mat{A}\mat{V}_k\mat{Y}
                      =\mat{Y}\mat{\Theta}$}}.
 
    \item If
          \alt<2>{$\|\vec{r}\,(=\mat{A}\vec{u}-\theta\vec{u})\|
                   \leq\mathrm{tol}$,
                  accept $(\theta,\vec{u})$}%
                 {\bl{$\|\vec{r}_i\,(=\mat{A}\vec{u}_i
                       -\theta_i\vec{u}_i)\|\leq\mathrm{tol}$
                  for $i=1,\ldots,b$,
                  accept $(\theta_i,\mat{u}_i)$}}
 
    \item Solve \textbf{correction equation} approximately:
    \[
      \alt<2>%
        {(\mat{I}-\vec{u}\vec{u}^\top)(\mat{A}-\theta\mat{I})
         (\mat{I}-\vec{u}\vec{u}^\top)\vec{t}=-\vec{r},
         \quad\vec{t}\perp\vec{u}}%
        {\bl{(\mat{I}-\mat{U}\mat{U}^\top)(\mat{A}-\theta_i \mat{I})
             (\mat{I}-\mat{U}\mat{U}^\top)\mat{t}_i=-\mat{r}_i,}
         \quad\bl{\mat{t}_i\perp\mat{U}}}
    \]
 
    \item
          \alt<2>{Orthogonalize $\vec{t}$ against $\mat{V}_k$
                  and normalize:}%
                 {\bl{Orthogonalize block $\mat{T} = [\vec{t}_1,...,\vec{t}_b]$ against $\mat{V}_k$,
                  then $\mat{T}\gets\mathrm{orth}(\mat{T})$:}}
    \[
      \alt<2>%
        {\vec{t}\gets\vec{t}-\mat{V}_k\mat{V}_k^\top\vec{t},
         \qquad\vec{t}\gets\vec{t}/\|\vec{t}\|_2}%
        {\bl{\mat{T}\gets\mat{T}-\mat{V}_k\mat{V}_k^\top\mat{T},
             \qquad\mat{T}\gets\mathrm{orth}(\mat{T})}}
    \]
 
    \item Expand subspace:
          $\mat{V}_{k+1}=\bigl[\mat{V}_k,\;
           \alt<2>{\vec{t}}{\bl{\mat{T}}}\bigr]$.
 
    \item Restart if $k$ too large.
 
  \end{enumerate}
\pause
\textbf{Remarks:} the inner solve (line 4) can be inexact (preconditioned
Krylov, low precision); the outer iteration provides refinement.
\end{frame}

\begin{frame}{JD correction equation}
\vspace{-0.3cm}
\begin{equation*}
  (\mat{I}-\vec{u}\vec{u}^\top)(\mat{A}-\theta\mat{I})
  (\mat{I}-\vec{u}\vec{u}^\top)\vec{t}=-\vec{r},
  \qquad\vec{t}\perp\vec{u}.
\end{equation*}
\vspace{-0.5cm}
\begin{itemize}
  \item Solving inexactly to tolerance proportional to
        $\|\mat{A}\vec{u}-\theta\vec{u}\|$ has the same effect as an exact solve.
  \item A cruder solve sacrifices convergence speed
        (quadratic $\rightarrow$ linear or \\ cubic $\rightarrow$ quadratic if Hermitian).
\end{itemize}
% \pause
% \vspace{0.3cm}
% \begin{equation*}
%     (\mat{I}-\vec{u}(\mat{S}\vec{u})^\top \mat{S})(\mat{A}-\tilde{\theta}\mat{I})
%     (\mat{I}-\vec{u}(\mat{S}\vec{u})^\top \mat{S})\vec{t}=-\vec{r},
%     \quad\vec{t}\perp_{\mat{S}}\vec{u}.
% \end{equation*}
% \vspace{-0.5cm}
% \begin{itemize}
%     \item $\tilde{\theta} = (\vec{Su})^\top(\vec{SAu})$ if $\mat{A}$ Hermitian
%     \item $\mat{\tilde{H}}_k = (\mat{SV}_k)^\top (\vec{SAV}_k)$ is not symmetric $\Rightarrow$ Convergence is quadratic if the (sketched) correction equation is solved exactly.
% \end{itemize}
\pause
\begin{block}{`Olsen' Solve}
    Solve correction equations via
    \begin{equation*}
        \vec{T} = -(\mat{I}-\mat{UU}^\top)\mat{K}^{-1}\vec{R} \quad \text{ or } \quad \vec{T} = -(\mat{I}-\mat{U}(\mat{SU})^\top\vec{S})\mat{K}^{-1}\vec{R}
    \end{equation*} where $\mat{K}$ is a preconditioner. No matvec with $A$, but still converges linearly!
\end{block}
\end{frame}

\begin{frame}{Sketched Galerkin Condition}
Change Galerkin condition to sketched Galerkin:
\begin{equation*}
  \mat{AV}\vec{y}-\lambda\mat{V}\vec{y}\perp\mat{V}
  \;\rightarrow\;
  \mat{AV}\vec{y}-\lambda\mat{V}\vec{y}\perp_{\mat{S}}\mat{V}
  \;\Rightarrow\;
  \mat{SAV}\vec{y}-\lambda\mat{SV}\vec{y}\perp\mat{SV}
\end{equation*}
\pause
leading to the reduced (generalized) eigenproblem
\begin{equation*}
  \mat{V}^\top\mat{AV}\vec{y}=\lambda\mat{V}^\top\mat{V}\vec{y}
  \;\rightarrow\;
  (\mat{SV})^\top(\mat{SAV})\vec{y}
  =\lambda(\mat{SV})^\top(\mat{SV})\vec{y}
\end{equation*}
\pause
\textbf{With RGS,} $(\mat{SV}_k)^\top\mat{SV}_k=\mat{I}$ by construction.
Substituting into the sketched Galerkin condition:
\begin{equation*}
  (\mat{SV}_k)^\top(\mat{SAV}_k)\vec{y}
  = \lambda\underbrace{(\mat{SV}_k)^\top(\mat{SV}_k)}_{=\,\mat{I}}\vec{y}
  \quad\Longrightarrow\quad
  \underbrace{(\mat{SV}_k)^\top(\mat{SAV}_k)}_{\tilde{\mat{H}}_k}\vec{y}
  = \lambda\vec{y}.
\end{equation*}
Standard (not generalized) EVP --- same structure as classical JD.

Downside: $\tilde{\mat{H}}_k$ is no longer symmetric.

\end{frame}

\begin{frame}{Jacobi-Davidson with sketched orthogonality}
  \textbf{Goal:} compute a few eigenpairs of
  $\mat{A}\in\R^{n\times n}$.
  \vspace{0.3em}

 
  \textbf{Input:} $\mat{A}$, target shift $\sigma$,
  \only<1-2>{initial vector $\vec{v}_1$}%
  \only<3->{\bl{initial block $\mat{V}_0\in\R^{n\times b}$}}.
 
  \begin{enumerate}
 
    \item Build\only<2->{ \rr{sketched}} orthogonal subspace
          \only<1>{$\mat{V}_k=[\vec{v}_1,\ldots,\vec{v}_k]$}
          \only<2>{\rr{$\tilde{\mat{V}}_k=[\vec{v}_1,\ldots,\vec{v}_k]$}}
          \only<2->{\rr{$(\mat{S\tilde{V}}_k)^\top\mat{S\tilde{V}}_k=\mat{I}$}}
 
    \item Compute
          \only<1>{Ritz pair $(\theta,\vec{u}=\mat{V}_k\vec{y})$}%
          \only<2>{Ritz pair $(\rr{\tilde{\theta},\tilde{\vec{u}}=\tilde{\mat{V}}_k \tilde{\vec{y}}})$}
          \only<3->{\bl{$(\tilde{\mat{\Theta}},\tilde{\mat{U}}=\tilde{\mat{V}}_k\tilde{\mat{Y}})$}}
          by solving
          \only<1>{$\mat{V}_k^\top\mat{A}\mat{V}_k\vec{y}=\theta\vec{y}$}%
          \only<2>{\rr{$(\mat{S\tilde{V}}_k)^\top(\mat{SA}\tilde{\mat{V}}_k)\tilde{\vec{y}}
                       =\tilde{\theta}\tilde{\vec{y}}$}}%
          \only<3->{\rr{$(\mat{S\tilde{V}}_k)^\top(\mat{SA}\tilde{\mat{V}}_k)$}%
                   \bl{$\tilde{\mat{Y}}=\tilde{\mat{Y}}\tilde{\mat{\Theta}}$}}
 
    \item If\only<2->{ \rr{sketched}} res.\ satisfies
          \only<1>{$\|\vec{r}\,(=\mat{A}\vec{u}-\theta\vec{u})\|
                   \leq\mathrm{tol}$}%
          \only<2>{\rr{$\|\mat{S}\mat{A}\tilde{\vec{u}}-\tilde{\theta}\mat{S}\tilde{\vec{u}}\|
                       \leq\mathrm{tol}$}}%
          \only<3->{\rr{$\|\mat{S}\mat{A}\tilde{\vec{u}}_i
                       -\tilde{\theta}_i\mat{S}\tilde{\vec{u}}_i\|\leq\mathrm{tol}$}%
                   \bl{, $i=1,\ldots,b$}},
          accept
          \only<1>{$\vec{u}$}%
          \only<2>{\rr{$\tilde{\vec{u}}$}}%
          \only<3->{\bl{$\tilde{\vec{u}}_i$}}
 
    \item Solve\only<2->{ \rr{sketched}} correction eq.\ approximately:
    \begin{equation*}
      \only<1>{%
        (\mat{I}-\vec{u}\vec{u}^\top)(\mat{A}-\theta\mat{I})
        (\mat{I}-\vec{u}\vec{u}^\top)\vec{t}=-\vec{r},
        \quad\vec{t}\perp\vec{u}}%
      \only<2>{\rr{%
        (\mat{I}-\tilde{\vec{u}}(\mat{S}\tilde{\vec{u}})^\top\mat{S})(\mat{A}-\tilde{\theta}\mat{I})
        (\mat{I}-\tilde{\vec{u}}(\mat{S}\tilde{\vec{u}})^\top\mat{S})\tilde{\vec{t}}=-\tilde{\vec{r}},
        \;\tilde{\vec{t}}\perp_{\!\mat{S}}\tilde{\vec{u}}}}%
      \only<3->{%
        \bl{(\mat{I}-\tilde{\vec{U}}(\mat{S}\tilde{\vec{U}})^\top\mat{S})
            (\mat{A}-\tilde{\theta}_i\mat{I})
            (\mat{I}-\tilde{\vec{U}}(\mat{S}\tilde{\vec{U}})^\top\mat{S})}%
        \bl{\tilde{\vec{t}}_i=-\tilde{\vec{r}}_i,}
        \;\bl{\forall i,\;\vec{t}_i\perp_{\!\mat{S}}\mat{U}}}
    \end{equation*}
 
    \item \only<1>{Orthogonalize $\vec{t}$ against $\mat{V}_k$}%
          \only<2>{\rr{Sketch-orthogonalize $\tilde{\vec{t}}$}}
          \only<3->{\rr{Sketch-orthogonalize} \bl{$\tilde{\vec{T}}$}}
          \only<2->{\rr{against $\tilde{\mat{V}}_k$}} and normalize:
    \begin{equation*}
      \only<1>{%
        \vec{t}\gets\vec{t}-\mat{V}_k\mat{V}_k^\top\vec{t},\quad
        \vec{t}\gets\vec{t}/\|\vec{t}\|_2}%
      \only<2>{\rr{%
        \tilde{\vec{t}}\gets\tilde{\vec{t}}-\tilde{\mat{V}}_k(\mat{S\tilde{V}}_k)^\top\mat{S}\tilde{\vec{t}},\quad
        \tilde{\vec{t}}\gets\tilde{\vec{t}}/\|\mat{S}\tilde{\vec{t}}\|_2}}%
      \only<3->{%
        \rr{\tilde{\mat{T}}\gets\tilde{\mat{T}}-\tilde{\mat{V}}_k(\mat{S\tilde{V}}_k)^\top\mat{S}\tilde{\mat{T}},}
        \quad\bl{\tilde{\mat{T}}\gets\mathrm{sketch\_orth}(\tilde{\mat{T}})}}
    \end{equation*}
 
    \item Expand:
          $\only<1>{\mat{V}_{k+1}=\bigl[\mat{V}_k,\;
           \vec{t}\bigr]}
           \only<2>{\rr{\tilde{\mat{V}}_{k+1}=\bigl[\tilde{\mat{V}}_k,\;\tilde{\vec{t}}\bigr]}}
           \only<3->{\bl{\tilde{\mat{V}}_{k+1}=\bigl[\tilde{\mat{V}}_k,\;\tilde{\mat{T}}\bigr]}}$.
 
    \item Restart if $k$ too large.
 
  \end{enumerate}
    \only<4>{ 
    \begin{itemize}
        \item block size: $b = k$ (OR $b = k + 2$ for robustness)
        \item restart sizes: $[2k, 5k]$ (OR $[3k, 6k]$ for robustness)
    \end{itemize}  
    }
\end{frame}
