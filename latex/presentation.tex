% --- Motivation & landscape ---
\begin{frame}{Motivation \& Context}
\begin{itemize}
  \item Compute selected eigenpairs of large sparse matrices: $\mat{Ax} = \lambda \vec{x}$
  \item Direct solvers are expensive, so we resort to iterative methods
  \pause
  \item \textbf{Challenges:} scalability, memory, communication, clustered/interior eigenvalues, conditioning, preconditioning, ...
  \item \textbf{Opportunities:}
    \begin{itemize}
      \item \textbf{Randomization}: projection onto lower-dimensional space for efficiency
      \item \textbf{Mixed-Precision}: fast low-precision bulk, high-precision critical steps.
    \end{itemize}
\end{itemize}
\pause
\begin{block}{Algorithms}
    \begin{itemize}
  \item \textbf{IRA} (Implicitly Restarted Arnoldi): Krylov + implicit restart; extremal eigenvalues.
  \item \textbf{JD} (Jacobi-Davidson Method): Rayleigh--Ritz + correction equation; good for interior/clustered eigenvalues.
  \item \textbf{LOBPCG} (Locally Optimal Block Preconditioned CG): Efficient for Hermitian/SPD problems.
\end{itemize}
\end{block}
\end{frame}


% --- IRA and rIRA ---
\begin{frame}{Implicitly Restarted Arnoldi (IRA)}
\begin{itemize}
  \item Arnoldi relation: $\mat{A V}_k = \mat{V}_{k}\mat{H}_k + h_{k+1,k}\vec{v}_{k+1}\vec{e}_{k}^\top$, $\mat{V}_k^\top\mat{V}_k = \mat{I}$.
  \item Implicit restart $\equiv$ polynomial filtering of unwanted spectral components.
\end{itemize}
\pause
\vspace{1em}
\textbf{Pros:}
\begin{itemize}
    \item Memory efficient: keeps Krylov subspace small
    \item Computationally efficient: filters out unwanted Ritz values
    \item Works well for extreme eigenvalues
    \item Preconditioner is not needed
\end{itemize}
\vspace{0.3cm}
\pause
\textbf{Cons:}
\begin{itemize}
    \item Convergence to interior eigenvalues can be very slow
    \item Convergence can be slow if eigenvalues are clustered
\end{itemize}
\vspace{1em}
\pause
\begin{itemize}
    \item \textbf{Randomized version:} {\footnotesize Balabanov-Grigori (2022); de Damas-Grigori (2024)}
    \item \textbf{Basic Idea:} Replace $\ell_2$-orthogonality with sketched orthogonality
\end{itemize}
\end{frame}

\begin{frame}{Randomization in Numerical Linear Algebra}
    \textbf{\underline{Basic Idea:}} 
\begin{itemize}
    \item We have a quantity of interest we would like to estimate, e.g., surrogate for $\mat{A}$ or row/column space of $\mat{A}$
    \item Use randomization/random matrices to approximate.
\end{itemize}
\vspace{1em}
\pause
\textbf{\underline{Pros and Cons:}} 
    \begin{itemize}
        \item[\bb{\textbf{+}}] Speed, Storage, Communication costs
        \item[\bb{\textbf{+}}] Robustness against adversarial examples
        \pause
        \item[\rr{\textbf{-}}] Output different for each run 
        \item[\rr{\textbf{-}}] Can fail with (very) small probability and lose a bit of accuracy
    \end{itemize}
    \pause
    \vspace{1em}
    \centering \large
    \textbf{Usually randomized algorithms obtain near-optimal solution very quickly!} \\
    {\footnotesize See Halko, Martinsson, Tropp, (2011); Woodruff, (2014); Drineas, Mahoney, (2017); Martinsson, Tropp, (2020);  Kireeva, Tropp (2024).}
\end{frame}

\begin{frame}{Subspace Embedding Property}
    \begin{block}{Subspace Embedding {\footnotesize [Sarl\'os, 2006]}}
    A matrix $\mat{S}\in \R^{s\times m}$ is a subspace embedding for $\mat{A}\in \R^{m\times n}$ if for all $\vec{x}\in \operatorname{range}(\mat{A})$ and for some distortion $\eta \in (0,1)$,
        \begin{equation*}
            (1-\eta)\norm{\vec{x}}_2^2 \leq \norm{\vec{Sx}}_2^2 \leq (1+\eta) \norm{\vec{x}}_2^2.
        \end{equation*} Typically, $s \ll m$ and $s = \bigO(\operatorname{rank}(\mat{A}))$.
        Analogues for $\mathbb{C}$ exist.
    \end{block}
    \textbf{Related topics:} Johnson-Lindenstrauss (JL) Lemma, Restricted Isometry Property (RIP).
    \pause
    \begin{block}{Examples of Random Embeddings}
        \begin{itemize}
            \item Scaled Gaussian Embedding ($S_{ij} \sim N(0,1/\sqrt{s})$)
            \item Subsampled Randomized Trigonometric Transforms (SRTTs)
            \item Sparse Sign Embedding
        \end{itemize}
    \end{block}
\end{frame}


\begin{frame}{Consequences:}
\begin{itemize}
    \item Inner product is approximately preserved:
    \begin{equation*}
        |\langle \mat{x},\mat{y}\rangle - \langle \mat{Sx},\mat{Sy}\rangle| \leq \eta \norm{\vec{x}}_2\norm{\vec{y}}_2
    \end{equation*} for all $\vec{x},\vec{y} \in \operatorname{range}(\mat{A})$.
    \pause
    \item Singular values are approximately preserved:
    \begin{equation*}
        (1-\eta)\sigma_{i}(\mat{A})^2 \leq \sigma_i(\mat{SA})^2 \leq (1+\eta)\sigma_i(\mat{A})^2
    \end{equation*}
    \pause
    \item Condition number is approximately similar:
    \begin{equation*}
        \kappa_2(\mat{SA}) \leq \sqrt{\frac{1+\eta}{1-\eta}} \cdot\kappa_2(\mat{A}), 
    \end{equation*} where $\kappa_2(\mat{A}) = \sigma_1(\mat{A})/\sigma_n(\mat{A})$.
\end{itemize}
\end{frame}

\begin{frame}{Sketched orthogonality}
Given basis $\mat{V}\in\mathbb{R}^{n\times k}$ and sketch $\mat{S}\in\mathbb{R}^{s\times n}$ ($s\ll n$, e.g. $s = 2k$):

\begin{itemize}
  \item Classical orthogonal projector (keep $\mat{V}$ orthogonal):
  \[
    \mat{P} = \mat{V}(\mat{V}^\top \mat{V})^{-1} \mat{V}^\top = \mat{VV}^\top,
  \]
  \pause
  \item Sketched / oblique projector (keep $\mat{SV}$ orthogonal):
  \[
    \mat{P}_{\mat{S}} = \mat{V}(\mat{SV})^{-1}\mat{S} = \mat{V}(\mat{SV})^\top \mat{S},
  \]
  where $\mat{P}_{\mat{S}}$ projects onto $\operatorname{span}(\mat{V})$ but orthogonality is defined in the \emph{sketched inner product} $\langle \vec{x},\vec{y}\rangle_{\mat{S}} = \langle \vec{Sx},\vec{Sy}\rangle$.
  \pause
  \item Keeping $\mat{SV}$ orthogonal $\Rightarrow$ well-conditioned basis $\kappa_2(\mat{V})\leq \sqrt{\frac{1+\eta}{1-\eta}}$
\end{itemize}
\end{frame}

% --- Sketched least squares and mixed precision ---
\begin{frame}{Sketched least squares formulation}
\small
\textbf{Classical GS step:} orthogonalize $\vec{t}$ against $\mat{V}$ ($\ell_2$-orthogonal) by solving
\[
  \min_{\vec{h}} \|\vec{t} - \vec{Vh}\|_2^2, \quad \text{solution: } \vec{h}_*=\vec{V}^\top \vec{t}.
\]
\pause
\textbf{Sketched GS step:} replace by sketched least squares
\[
  \min_{\vec{h}} \|\vec{S}(\vec{t} - \vec{Vh})\|_2^2, \quad \text{solution: } \vec{\tilde{h}}_*=(\mat{SV})^\dagger \mat{S} \vec{t}.
\]
\pause
\begin{itemize}
  \item If $\mat{S}$ is a subspace embedding for $[\mat{V},\vec{t}]$
  \begin{equation*}
      \|\vec{t} - \vec{V\tilde{h}}_*\|_2^2 \leq \frac{1+\eta}{1-\eta} \|\vec{t} - \vec{Vh}_*\|_2^2.
  \end{equation*}
  \item $\Rightarrow$ Orthonormalization, Rayleigh--Ritz, and residual checks can all be phrased in this sketched LS framework.
  \pause
  \item \textbf{Mixed precision is natural:}
    \begin{itemize}
      \item Forming $\mat{SV}$ and $\vec{St}$: high-dimensional computation in low-precision
      \item Solving small $k\times k$ systems/eigenvalue problem: lower-dimensional computation in high-precision.
    \end{itemize}
\end{itemize}
\end{frame}


% --- JD algorithm: overview ---
\begin{frame}{Jacobi--Davidson algorithm (overview)}
\textbf{Goal:} compute a few eigenpairs of $\mat{A}\in\mathbb{R}^{n\times n}$. \\
\vspace{0.5em}
\pause
\textbf{Input:} $\mat{A}$, target shift $\sigma$, initial vector $\vec{v}_1$.

\begin{enumerate}
  \item Build orthogonal subspace $\vec{V}_k = [\vec{v}_1,\ldots,\vec{v}_k]$.
  \item Compute Ritz pair $(\theta,\vec{u}=\vec{V}_k\vec{y})$ by solving $\mat{V}_k^\top \mat{AV}_k\vec{y} = \theta \vec{y}$.
  \item If residual satisfies $\|\vec{r} (= \vec{Au}-\theta \vec{u})\|\le \text{tol}$, accept $(\theta,u)$ (lock/deflate).
  \item Solve \textbf{correction equation} approximately:
  \[
    (\mat{I}-\vec{uu}^\top)(\mat{A}-\theta \mat{I})(\mat{I}-\vec{uu}^\top)\vec{t} = -\vec{r},\quad \vec{t}\perp \vec{u}.
  \]
  \item Orthogonalize $\vec{t}$ against $\mat{V}_k$ and normalize:
  \begin{equation*}
      \vec{t} \gets \vec{t} - \mat{V}_k\mat{V}_k^\top\vec{t},\quad \vec{t}\gets \vec{t}/\|\vec{t}\|_2
  \end{equation*}
  \item Expand subspace: $\mat{V}_{k+1} = [\mat{V}_k,\; \vec{t}]$.
  \item Restart if $k$ too large (thick restart, keep best Ritz vectors).
\end{enumerate}
\pause
\textbf{Remarks:} 

Inner solve (line $4$) can be inexact (preconditioned Krylov, low precision); outer iteration provides refinement.

With block form: replace $(\vec{u},\theta)$ by Ritz block $(\mat{U},\mat{\Theta})$ and $\vec{t}$ by block $\mat{T}$.
\end{frame}

% --- Derivation of correction equation ---
\begin{frame}{Derivation of JD correction equation}
Given current Ritz pair $(\theta,\vec{u})$ we seek a correction $\vec{t}$ orthogonal to $\vec{u}$:
\begin{align*}
    \mat{A}(\vec{u}+\vec{t}) &= \lambda(\vec{u}+\vec{t}), \quad \vec{u}\perp\vec{t}
\end{align*}
\pause
Multiply $\mat{I}-\vec{uu}^\top$ and rearranging gives:
\begin{equation*}
    (\mat{I}-\vec{uu}^\top)\mat{A}(\vec{u}+\vec{t}) =\lambda \vec{t} \implies (\mat{I}-\vec{uu}^\top)(\mat{A}-\lambda\mat{I}) \vec{t} = -(\mat{Au}-\theta \vec{u}) (= -\vec{r}).
\end{equation*} \pause Now $(\mat{I}-\vec{uu}^\top)\vec{t} = \vec{t}$ and approximating the eigenvalue $\lambda$ with Ritz value $\theta$ gives the JD correction equation:
\begin{equation} \label{eq:JDCE}
    (\mat{I}-\vec{uu}^\top)(\mat{A}-\theta \mat{I})(\mat{I}-\vec{uu}^\top)\vec{t} = -\vec{r}.
\end{equation} Note that $\vec{u} \perp \vec{t}$ is needed as $\vec{t} = \vec{u}$ is a solution to \eqref{eq:JDCE}, otherwise leading to stagnation. \pause
\begin{itemize}
    \item Solving JD correction equation exactly: same convergence rate with Rayleigh Quotient Iteration (Quadratic local convergence)
    \item Solving JD correction equation exactly with a fixed shift instead of Ritz value $\theta$: Shifted inverse iteration (Linear convergence)
\end{itemize}
\end{frame}

\begin{frame}{Jacobi-Davidson with sketched orthogonality}
    Change Galerkin condition to Sketched Galerkin condition:
  \begin{equation*}
      \mat{AV}\vec{y} - \lambda \vec{Vy} \perp \mat{V} \Rightarrow \mat{AV}\vec{y} - \lambda \vec{Vy} \perp_{\mat{S}} \mat{V} \Rightarrow \mat{SAV}\vec{y} - \lambda \vec{SVy} \perp \mat{SV} 
  \end{equation*} \pause leading to the reduced eigenproblem
  \begin{align*}
      \mat{V}^\top\mat{AV}\vec{y} = \lambda \mat{V}^\top\mat{V}\vec{y} \Rightarrow (\mat{SV})^\top (\mat{SAV}) \vec{y} = \lambda (\mat{SV})^\top (\mat{SV})\vec{y} 
  \end{align*}
\pause
\vspace{-0.5cm}
\begin{enumerate}
  \item Build\only<4>{ \rr{sketched}} orthogonal subspace $\vec{V}_k = [\vec{v}_1,\ldots,\vec{v}_k]$ \only<4>{\rr{$(\mat{SV}_k)^\top \mat{SV}_k = \mat{I}$}}
  \item Compute Ritz pair $(\theta,\vec{u}=\vec{V}_k\vec{y})$ by solving \only<3>{$\mat{V}_k^\top \mat{AV}_k\vec{y} = \theta \vec{y}$}\only<4>{\rr{$(\mat{S}\mat{V}_k)^\top (\mat{SAV}_k)\vec{y} = \theta \vec{y}$}}
  \item If \only<4>{\rr{sketched} }res. satisfies \only<3>{$\|\vec{r} (= \vec{Au}-\theta \vec{u})\|\le \text{tol}$}\only<4>{\rr{$\|\vec{SAu}-\theta \vec{Su}\|\le \text{tol}$}}, accept $\vec{u}$ (lock/deflate)
  \item Solve \textbf{correction equation} approximately: \only<4>{\rr{(next slide)}}
  \[
    (\mat{I}-\vec{uu}^\top)(\mat{A}-\theta \mat{I})(\mat{I}-\vec{uu}^\top)\vec{t} = -\vec{r},\quad \vec{t}\perp \vec{u}.
  \]
  \item \only<3>{Orthogonalize}\only<4>{\rr{Sketch orthogonalize}} $\vec{t}$ against $\mat{V}_k$ and normalize:
  \begin{equation*}
      \only<3>{\vec{t} \gets \vec{t} - \mat{V}_k\mat{V}_k^\top\vec{t},\quad \vec{t}\gets \vec{t}/\|\vec{t}\|_2}
      \only<4>{\rr{\vec{t} \gets \vec{t} - \mat{V}_k(\mat{SV}_k)^\top\mat{S}\vec{t},\quad \vec{t}\gets \vec{t}/\|\vec{St}\|_2}}
  \end{equation*}
  \item Expand subspace: $\mat{V}_{k+1} = [\mat{V}_k,\; \vec{t}]$.
  \item Restart if $k$ too large (thick restart, keep best Ritz vectors).
\end{enumerate}
\end{frame}

% --- Inexact solves & low precision ---
\begin{frame}{JD Correction equation}
\begin{equation*}
    (\mat{I}-\vec{uu}^\top)(\mat{A}-\theta \mat{I})(\mat{I}-\vec{uu}^\top)\vec{t} = -\vec{r},\quad \vec{t}\perp \vec{u}.
\end{equation*}
\begin{itemize}
    \item Solving the correction equation inexactly to the tolerance proportional to the residual $\|\mat{Au}-\theta\vec{u}\|$ has the same effect as exact solve.
    \item Can be more inexact but sacrifices convergence speed. (Quadratic $\rightarrow$ Linear)
\end{itemize}
 \pause
\underline{Some considerations:}
\begin{itemize}
    \item Low-precision can be used
    \item Sketched orthogonal projector $\mat{I} - \mat{u}(\mat{Su})^\dagger \mat{S}$ instead of orthogonal projector $\mat{I}-\vec{uu}^\top$.
    \item sGMRES with truncated Arnoldi can be used; {\footnotesize Nakatsukasa, Tropp (2024)}
    \item Replace $\mat{A}$ by a preconditioner $\mat{K}$:
    \begin{equation*}
    (\mat{I}-\vec{uu}^\top)(\mat{K}-\theta \mat{I})(\mat{I}-\vec{uu}^\top)\vec{t} = -\vec{r}
\end{equation*}
\end{itemize}
\end{frame}


% ===== LOBPCG: Algorithm, Theory, Mixed Precision, Randomization =====

% --- Problem + search subspace ---
\begin{frame}{LOBPCG}
\textbf{Aim:} Solve Generalized Hermition EVP: 
\begin{equation*}
    \mat{AV} = \mat{B V}\mat{\Lambda}, \quad \mat{A},\mat{B} \text{ Hermitian}, \,\,\,\mat{B} \succ 0
\end{equation*}
\pause
At each iteration finds, locally optimal subspace:
\begin{equation*}
    \mathcal{S}_k \;=\; \mathrm{span}\{\,\mat{V}_k,\; \mat{W}_k,\; \mat{V}_{k-1}\,\}, \,\,\, \mat{R}_k := \mat{A V}_k - \mat{B V}_k \mat{\Theta}_k, \,\,\, \mat{W}_k := \mat{K R}_k
\end{equation*} with Ritz pairs $(\mat{\Theta}_k,\mat{V}_k)$ and precond. $\mat{K}$. \pause Then takes locally optimal step: $\mat{V}_{k+1}$ is a $\mat{B}$-orthonormal Ritz block minimizing the Rayleigh quotient over $\mathcal{S}_k$. ($(3b)\!\times\! (3b)$ Rayleigh-Ritz solve, block size $b$) \\
\vspace{0.5em}
\pause
\textbf{\underline{Comments:}}
\begin{itemize}
    \item Preconditioned Eigensolvers; {\footnotesize Neymeyr (2001); Knyazaev (2003)}
    \item Local Optimality: $\mat{V}_{k+1}$ minimizes $\operatorname{trace}(\mat{V}^\top \mat{A V})$ over
$\{\mat{V}\in\mathcal{S}_k : \mat{V}^\top \mat{B V} = \mat{I}\}$ $\Rightarrow$ monotone Ritz values
    \item Weak theory. Local convergence to some eigenpair at linear rate.
    \item Works very well in practice.
    \item Hard/soft lock converged Ritz pairs.
\end{itemize}
\end{frame}

% --- Mixed precision policy (aligned with Kressner--Ma--Shao) ---
\begin{frame}{Mixed precision in LOBPCG}
\textbf{Fast path in lower precision:}
\begin{itemize}
  \item Apply preconditioner: $\mat{W}_k = \mat{K R}_k$ (often inherently low-accuracy).
\end{itemize}
\pause
\textbf{Accuracy-critical in high precision:}
\begin{itemize}
  \item Small $(3b)\!\times\!(3b)$ Rayleigh--Ritz solve.
  \item $\mat{B}$-orthonormalization of $\mat{V}_{k+1}$; residual norms for locking/deflation.
\end{itemize}
\pause
\textbf{Outcome:} preserves accuracy and convergence while typically yielding $1.4$–$2\times$ speedups on modern CPU/GPU systems; Kressner, Ma, Shao (2023).
\pause
\begin{block}{Randomization?}
    \begin{itemize}
    \item For standard EVP ($\mat{B} = \mat{I}$), sketched inner product as before.
    \item For generalized EVP, tricky as the obvious way is 
    \begin{equation*}
        \langle \vec{x},\vec{y} \rangle_{\mat{B}} = \langle \mat{B}^{1/2} \vec{x}, \mat{B}^{1/2}\vec{y} \rangle \Rightarrow  \langle \mat{S}\mat{B}^{1/2} \vec{x}, \mat{S}\mat{B}^{1/2}\vec{y} \rangle, 
    \end{equation*} which requires cholesky/square root of $\mat{B}$. We can try
        $\langle \mat{S} \vec{x}, \mat{SB}\vec{y} \rangle$, but disrespects symmetry in the problem.
\end{itemize}
\end{block} 
\end{frame}

% --- Conclusion slide ---
\begin{frame}{Conclusions}
\begin{itemize}
  \item \textbf{Randomization} (sketched orthogonality):
    \begin{itemize}
      \item Reduces communication bottlenecks %($\mat{V}^\top \vec{t}$ vs $(\mat{SV})^\top \mat{St}$).
      \item Preserves subspace geometry up to distortion $O(\eta)$.
      \item Achieves speedups with little accuracy loss.
    \end{itemize}
    \pause
  \item \textbf{Mixed precision}:
    \begin{itemize}
      \item Natural for sketching, inexact inner solves, and preconditioning.
      \item Critical steps (Rayleigh--Ritz, residual norms) remain in higher precision.
      \item Achieves speedups with accuracy guarantees.
    \end{itemize}
\end{itemize}
\pause
\vspace{2mm}
\centering
Randomization + mixed precision can be incorporated into existing eigensolvers to provide mathematically justified, hardware-aware strategies that make eigensolvers scalable.
\end{frame}
